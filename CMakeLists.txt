cmake_minimum_required(VERSION 3.25)
cmake_policy(SET CMP0141 NEW)
set(CMAKE_MSVC_DEBUG_INFORMATION_FORMAT Embedded)
project(Botzin VERSION 0.1.0 LANGUAGES C RC)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "Botzin requires Windows x86")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_library(botzin_format STATIC src/botzin_sparse.c)
target_include_directories(botzin_format PUBLIC include)
add_library(botzin_diagnostics STATIC src/botzin_diagnostics.c)
target_include_directories(botzin_diagnostics PUBLIC include)
add_library(botzin_host SHARED src/botzin_host.c resources/botzin_host.rc)
target_include_directories(botzin_host PRIVATE include resources)
target_link_libraries(botzin_host PRIVATE user32 comctl32 botzin_diagnostics)
target_link_options(botzin_host PRIVATE /MANIFEST:NO)
add_executable(sparse_test tests/sparse_test.c)
target_link_libraries(sparse_test PRIVATE botzin_format)
add_executable(abi_test tests/abi_test.c)
target_include_directories(abi_test PRIVATE include)
add_executable(diagnostics_test tests/diagnostics_test.c)
target_link_libraries(diagnostics_test PRIVATE botzin_diagnostics)
foreach(target botzin_format botzin_diagnostics botzin_host sparse_test abi_test diagnostics_test)
  target_compile_options(${target} PRIVATE /W4 /WX /fp:precise)
  target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
endforeach()
foreach(target botzin_host sparse_test abi_test diagnostics_test)
  target_link_options(${target} PRIVATE /DEBUG:FULL "/PDBALTPATH:%_PDB%" "$<$<CONFIG:Release>:/OPT:REF>" "$<$<CONFIG:Release>:/OPT:ICF>")
  install(FILES "$<TARGET_PDB_FILE:${target}>" DESTINATION symbols OPTIONAL)
endforeach()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
enable_testing()
add_test(NAME sparse-profiles COMMAND sparse_test "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
add_test(NAME initialization-abi COMMAND abi_test)
add_test(NAME runtime-diagnostics COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/check_runtime_log.py" "$<TARGET_FILE:diagnostics_test>")
add_test(NAME compiler-diagnostics COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_diagnostics.py")
install(TARGETS botzin_format botzin_diagnostics botzin_host sparse_test abi_test diagnostics_test RUNTIME DESTINATION bin LIBRARY DESTINATION bin ARCHIVE DESTINATION lib)
